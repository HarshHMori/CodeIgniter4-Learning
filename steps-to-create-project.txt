-- check composer is installed by typing "composer"

-- create project by this
    "composer create-project codeigniter4/appstarter url-shortener"

-- check spark is installed
    "php spark"

-- making controller (URLController)
    "php spark make:controller URL --suffix

-- to start development server
    "php spark serve"
